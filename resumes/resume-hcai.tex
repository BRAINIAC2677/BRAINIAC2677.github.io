\newcommand{\pulsarpublication}[1]{%

    {PULSAR: Graph-Based Positive Unlabeled Learning with Multi-Stream Adaptive Convolutions for Parkinson's Disease Recognition}
    {Apr 2026}
    {\textit{Md Zarif Ul Alam\textsuperscript{*}, \textbf{\underline{Asif Azad}}\textsuperscript{*}, Md Saiful Islam, Ehsan Hoque, M Saifur Rahman} \enspace \textcolor{mutedColor}{\textsuperscript{*}Equal contribution}}

}

\newcommand{\vlmpublication}[1]{%
\selectedpublication
    {#1}
    {The Art of Saying ‘Maybe’: A Conformal Lens for Uncertainty Benchmarking in VLMs}
    {Mar 2026}
    {\textit{\textbf{\underline{Asif Azad}}, Mohammad Sadat Hossain, MD Sadik Hossain Shanto, M Saifur Rahman, Md Rizwan Parvez}}
    {\textbf{Findings of EACL 2026} \enspace\textbar\enspace \textbf{EACL: ICORE A} \enspace\textbar\enspace Published \enspace\textbar\enspace \href{https://aclanthology.org/2026.findings-eacl.274/}{\underline{Paper Link}}}
    {Established a \textbf{comprehensive conformal uncertainty benchmark} of \textbf{18 open- and closed-source VLMs across six multimodal datasets and three scoring functions}, including likelihood proxies for API-only models. The study shows that stronger models generally quantify uncertainty better, while uncertainty estimation remains \textbf{particularly challenging on mathematical and reasoning-intensive tasks}.}
}

    {Aug 2025}
    {\textit{\textbf{\underline{Asif Azad}}, Md. Saiful Islam, Ehsan Hoque, M. Saifur Rahman}}
    {\textbf{AIiH 2025, Springer LNCS} \enspace\textbar\enspace Published \enspace\textbar\enspace \textbf{Best Paper Award Nomination} \enspace\textbar\enspace \href{https://link.springer.com/chapter/10.1007/978-3-032-00652-3_3}{\underline{Paper Link}} \enspace\textbar\enspace \href{https://github.com/BRAINIAC2677/UQ4PD-ML}{\underline{Code}}}
