\section{Selected Publications}

\selectedpublication
    {1}
    {Harf-Speech: A Clinically Aligned Framework for Arabic Phoneme-Level Speech Assessment}
    {Sep 2026}
    {\textit{\textbf{\underline{Asif Azad}}, MD Sadik Hossain Shanto, Mohammad Sadat Hossain, Bdour Alwuqaysi, Sabri Boughorbel, Yahya Bokhari, Abdulrhman Aljouie, Ayah Othman Sindi, Ehsan Hoque}}
    {\href{https://www.isca-speech.org/event-5462701}{\textbf{Interspeech 2026}} (accepted; proceedings forthcoming) \enspace\textbar\enspace \href{https://arxiv.org/abs/2604.06191}{\underline{Paper Link}}}

It achieved an \textbf{8.92\% phoneme error rate} and \textbf{strong agreement with speech-language pathologists (\(r=0.791\))}, enabling scalable assessment for an underserved language.}

\selectedpublication
    {2}
    {Digital Harf: A Clinically Integrated Multimodal AI System for Pervasive Arabic Speech and Language Therapy}
    {May 2026}
    {\textit{\textbf{\underline{Asif Azad}}, Mohammad Sadat Hossain, MD Sadik Hossain Shanto, Sabri Boughorbel, Abdulrhman Aljouie, Bdour Alwuqaysi, Yahya Bokhari, Ayah Othman Sindi, Ehsan Hoque}}
    {\textbf{Preprint} \enspace\textbar\enspace \href{https://arxiv.org/abs/2607.27212}{\underline{Paper Link}} \enspace\textbar\enspace \href{https://digitalharf.modhs.med.sa/}{\underline{Project Link}}}
    {Built a \textbf{clinically integrated multimodal AI platform} for home-based Arabic speech and language assessment and intervention. Its \textbf{agentic synthetic-data engine} generated culturally grounded therapeutic content with \textbf{90.1\% clinician acceptance}; the platform is being evaluated in an \textbf{ongoing randomized clinical trial}.}

\selectedpublication
    {3}
    {Validation of Remote Multimodal AI Screening for Parkinson Disease Across Diverse Settings}
    {May 2026}
    {\textit{Md Saiful Islam, Tariq Adnan, Abdelrahman Abdelkader, Zipei Liu, Evelyn Ma, Sooyong Park, \textbf{\underline{Asif Azad}}, Pai Liu, Meghan Pawlik, Emily Hartman, Erin Shelton, Kristina B. Larson, M. Saifur Rahman, Cathe Schwartz, Karen Jaffe, Jamie L. Adams, Ruth B. Schneider, Jan Freyberg, E. Ray Dorsey, Ehsan Hoque}}
    {\href{https://www.nature.com/articles/s43856-026-01606-6}{\textbf{Communications Medicine}} (published; open access) \enspace\textbar\enspace \href{https://www.nature.com/articles/s43856-026-01606-6}{\underline{Paper Link}}}

\selectedpublication
    {4}
    {PULSAR: Graph-Based Positive Unlabeled Learning with Multi-Stream Adaptive Convolutions for Parkinson's Disease Recognition}
    {Apr 2026}
    {\textit{Md Zarif Ul Alam, \textbf{\underline{Asif Azad}}, Md Saiful Islam, Ehsan Hoque, M Saifur Rahman}}
    {\href{https://dl.acm.org/doi/10.1145/3799417}{\textbf{ACM Transactions on Computing for Healthcare}} (published) \enspace\textbar\enspace \href{https://dl.acm.org/doi/10.1145/3799417}{\underline{Paper Link}}}
    {Co-developed \textbf{PULSAR, a positive-unlabeled multi-stream graph-convolutional framework} for Parkinson's screening from webcam finger-tapping videos. By modeling noisy self-reported labels and joint, bone, velocity, and acceleration streams, it achieved \textbf{71.3\% mean accuracy on an independent test set} despite limited reliable labels.}

\selectedpublication
    {5}
    {The Art of Saying ‘Maybe’: A Conformal Lens for Uncertainty Benchmarking in VLMs}
    {Mar 2026}
    {\textit{\textbf{\underline{Asif Azad}}, Mohammad Sadat Hossain, MD Sadik Hossain Shanto, M Saifur Rahman, Md Rizwan Parvez}}
    {\href{https://aclanthology.org/2026.findings-eacl.274/}{\textbf{Findings of EACL 2026}} (published; peer reviewed) \enspace\textbar\enspace \href{https://aclanthology.org/2026.findings-eacl.274/}{\underline{Paper Link}}}
    {Established a \textbf{comprehensive conformal uncertainty benchmark} of \textbf{18 open- and closed-source VLMs across six multimodal datasets and three scoring functions}, including likelihood proxies for API-only models. The study shows that stronger models generally quantify uncertainty better, while uncertainty estimation remains \textbf{particularly challenging on mathematical and reasoning-intensive tasks}.}

\selectedpublication
    {6}
    {Beyond Accuracy: Enhancing Parkinson's Diagnosis with Uncertainty Quantification of Machine Learning Models}
    {Aug 2025}
    {\textit{\textbf{\underline{Asif Azad}}, Md. Saiful Islam, Ehsan Hoque, M. Saifur Rahman}}
    {\href{https://link.springer.com/chapter/10.1007/978-3-032-00652-3_3}{\textbf{AIiH 2025, Springer LNCS}} (published; Best Paper Award nominee) \enspace\textbar\enspace \href{https://link.springer.com/chapter/10.1007/978-3-032-00652-3_3}{\underline{Paper Link}} \enspace\textbar\enspace \href{https://github.com/BRAINIAC2677/UQ4PD-ML}{\underline{Code}}}
    {Evaluated \textbf{Monte Carlo Dropout (MCD), Deep Evidential Classification (DEC), and Bayesian Neural Networks (BNNs)} across motor, facial, and speech datasets for Parkinson's diagnosis. The study found \textbf{MCD and BNNs more reliable than DEC} and showed how uncertainty estimates can \textbf{flag ambiguous predictions} to support safer clinical decision-making.}
